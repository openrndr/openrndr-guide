{# Navigation components. They use just-the-docs class names so its CSS applies. Pages and subsections of a section are merged and sorted by `weight` (taken from the number prefix of the .kt file / folder names), so a subsection can appear anywhere between pages. Works at any depth. {%- endif -%} {% endcomponent nav.children %} {# A section with an expander button and its children #} {% component nav.section(path: string, current: string) %} {%- set s = get_section(path=path, metadata_only=true) -%} {%- set has_children = s.pages | length > 0 or s.subsections | length > 0 -%} {%- set open = current is starting_with(pat=s.path) -%}
